\section{项目经历}

\projectheading
  {基于 PatchX-MLP 的 SCR 入口 NOx 浓度预测}
  {2025.03--2025.08}

\begin{itemize}
  \item \textbf{项目简介：}
  基于多变量工业运行数据预测 SCR 入口 NOx 浓度，
  为脱硝过程运行优化提供依据。
  \item \textbf{技术栈：}
  Python、PyTorch、MLP、多尺度 Patch、时序分解、时频域联合损失。
  \item \textbf{模型设计：}
  融合多尺度 Patch 与趋势、季节项分解，
  建模 Patch 内、Patch 间及变量间交互，
  通过时域误差与频域约束联合优化。
  \item \textbf{实验结果：}
  完成对比与消融实验；工业数据 96 步预测的
  MSE/MAE 为 \textbf{0.228/0.223}，
  较 TimeMixer 分别降低 \textbf{6.6\%/5.5\%}。
\end{itemize}

\projectheading
  {基于大模型的工作票智能生成系统}
  {2025.09--2026.09}

\begin{itemize}
  \item \textbf{项目简介：}
  编排图纸解析、知识检索、内容生成与校验流程，
  辅助生成电气检修工作票中的工作内容和安全措施。
  \item \textbf{技术栈：}
  Python、Dify、LLM Agent、Tool Calling、
  VL Embedding、SAM、OCR、GraphRAG。
  \item \textbf{图纸解析：}
  匹配图例与多尺度图块的视觉特征，定位设备候选区域，
  结合分割、文字识别与位置约束，提取设备类别、位置及编号。
  \item \textbf{检索与生成：}
  结合实体关系检索与文本语义召回，关联设备、检修任务、
  安规条款和历史工作票；将解析与检索封装为工具，
  基于来源证据和结构化模板生成内容，
  设置必填项、设备一致性检查及人工复核。
  \item \textbf{领域适配：}
  设计历史工作票指令样本与 LoRA 微调方案，
  以“检修任务＋检索证据”为输入、票面字段为输出，
  规划术语一致性、字段完整性及证据符合性评估。
\end{itemize}

\projectheading
  {大模型话题与安全检测系统}
  {2026.08--2026.10}

\begin{itemize}
  \item \textbf{项目简介：}
  构建文本与多语种音频检测模块，识别业务话题、
  提示词注入及有害内容，为请求放行、拦截与复核提供依据。
  \item \textbf{技术栈：}
  Python、PyTorch、Transformers、FastAPI、
  多任务学习、SFT、LoRA、CosyVoice。
  \item \textbf{文本检测：}
  基于预训练编码器构建话题、攻击与有害内容分类头，
  通过掩码损失利用多来源数据的部分标签；
  完成数据清洗与数据集划分，按误报约束校准阈值，
  结合白名单与风险得分制定决策策略。
  \item \textbf{音频检测：}
  监督微调多语种音频安全检测模型，识别输入音频中的
  有害内容与提示词注入，以及输出音频中的有害内容；
  通过语音合成生成训练样本，并添加噪声进行数据增强。
  \item \textbf{工程实现：}
  封装训练、评估与推理流程，提供文本检测接口，
  返回类别、分数与触发原因，
  对低置信度或检测能力缺失的请求设置复核分支。
\end{itemize}
